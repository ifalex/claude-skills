{{- else if eq .Values.service.type "ClusterIP" }}

  The Service is ClusterIP (internal). Port-forward to reach it locally:

    export SVC_PORT=$(kubectl get svc --namespace {{ include "CHARTNAME.namespace" . }} {{ include "CHARTNAME.fullname" . }} -o jsonpath='{.spec.ports[0].port}')
    kubectl port-forward --namespace {{ include "CHARTNAME.namespace" . }} svc/{{ include "CHARTNAME.fullname" . }} 8080:${SVC_PORT}
    echo "Visit http://127.0.0.1:8080"

{{- else if eq .Values.service.type "NodePort" }}

    export NODE_PORT=$(kubectl get svc --namespace {{ include "CHARTNAME.namespace" . }} {{ include "CHARTNAME.fullname" . }} -o jsonpath='{.spec.ports[0].nodePort}')
    export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[0].address}')
    echo "Visit http://${NODE_IP}:${NODE_PORT}"

{{- else if eq .Values.service.type "LoadBalancer" }}

  NOTE: the LoadBalancer IP may take a minute to provision.
    Watch: kubectl get svc --namespace {{ include "CHARTNAME.namespace" . }} -w {{ include "CHARTNAME.fullname" . }}

    export SVC_IP=$(kubectl get svc --namespace {{ include "CHARTNAME.namespace" . }} {{ include "CHARTNAME.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
    export SVC_PORT=$(kubectl get svc --namespace {{ include "CHARTNAME.namespace" . }} {{ include "CHARTNAME.fullname" . }} -o jsonpath='{.spec.ports[0].port}')
    echo "Visit http://${SVC_IP}:${SVC_PORT}"
